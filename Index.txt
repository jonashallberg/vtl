/*  1. BESKRIVNING AV BERÄKNINGEN 
    
    För att kunna göra bra omvärderingsberäkningar för in- och upplåning samt 
    utlåning är det viktigt att ta reda på valutautvecklingen för varje MFI-sektor. 
    Beräkningsfilen räknar fram viktade valutautvecklingar för respektive MFI-sektor. 
    Resultatet används därefter som indata i beräkningsfilen för valutakursomvärdering.  */

/*  2.0 IMPORT AV DATA & DATA-ANPASSNING  */   

    Valuta
            :=
                union(
                        Valuta_0
                        
                        ,Valuta_Tmin
                )
                [rename INS_Varde to NR_Varde]
                                                                                                                        ;

/*  Filtrerar ut ett dataset med DKK & GBP och ett med CHF, JPY, OVR och USD */

    FIR_index_1      
            :=      
                ratio_to_report(
                                    union(
                    
                                            Valuta
                                            [filter NR_Referenssektor   in  {"S12211"
                                                                            ,"S1223"
                                                                            ,"S12241"
                                                                            ,"S12242"}  and 
                                                                     
                                                    NR_Valuta           in  {"DKK"
                                                                            ,"GBP"}     and
                
                                                    cast(substr(NR_Referensperiod,1,4),integer) < 2020]                                
                
                                            ,Valuta
                                            [filter NR_Referenssektor   in  {"S12211"
                                                                             ,"S1223"
                                                                             ,"S12241"
                                                                             ,"S12242"} and 
                                                                     
                                                    NR_Valuta           in  {"DKK"}     and
                
                                                    cast(substr(NR_Referensperiod,1,4),integer) > 2019]                    
                                    )
                                /* Ratio to report funktionen skapar andelar som sedan används som vikter. DKK & GBP
                                är en grupp där valuta-utvecklingen bestäms via viktning av DKK och GBP's relativa
                                vikter. Likaså görs i gruppen CHF, JPY, OVR, USD */
                                over (
                                        partition by    NR_Referenssektor
                                                        ,NR_Bokforingspost
                                                        ,NR_Referensperiod
                                )
                )                                                                               
                                                                                                                        ;
                                                                    
    FIR_index_2      
            :=  
                ratio_to_report(
                                    Valuta
                                    [filter NR_Referenssektor   in  {"S12211"
                                                                    ,"S1223"
                                                                    ,"S12241"
                                                                    ,"S12242"}  and 
                                                                     
                                            NR_Valuta           in  {"CHF" 
                                                                    ,"JPY" 
                                                                    ,"OVR" 
                                                                    ,"USD"
                                                                    ,"GBP"}]                    
                                                                     
                                over (
                                        partition by    NR_Referenssektor
                                                        ,NR_Bokforingspost
                                                        ,NR_Referensperiod
                                )
                )                                     
                                                                                                                        ;
//region
/*  3.0 BERÄKNING  */   

/*  Nedan multiplicerar jag respektive andel (framräknat med ratio to report) med valutautvecklingen 
    för varje valuta. Sedan summerar jag alla rader i datasetet till en grupp av valutor, 
    VU2 (övriga eu), VU3 (övriga världen) och VU4 (euro), som jag ger en ny komponent: NR_Valuta */

    FIR_index_VU3      
            :=  
                sum(
                        FIR_index_2 * Index                                                                             
                
                    group by    NR_Bokforingspost
                                ,NR_Referenssektor
                                ,NR_Referensperiod
                )                                      
                [calc identifier NR_Valuta := "VU3"]                      
                                                                                                                        ;
                
    /*  Samma som ovan för gruppen CHF, JPY, OVR, och USD */
    
    FIR_index_VU2      
            :=  
                sum(
                        FIR_index_1 * Index                                                                              
    
                    group by    NR_Bokforingspost
                                ,NR_Referenssektor
                                ,NR_Referensperiod
                )
                [calc identifier NR_Valuta := "VU2"]
                                                                                                                        ;
                
    /*  Tar fram valutautvecklingen för Euro att lägga till de andra dataseten. Här behöver jag inte 
        göra några viktningar eller andra förändringar. Vad jag gör är att utgå ifrån ett dataset som 
        har en rad för varje sektor och bokföringspost. Sedan skapar jag en ny identifierare med VU4
        och ger alla rader Euron's utveckling  */
    
    FIR_index_VU4      
            :=  
                sum(
                        FIR_index_VU2                     
                
                    group by    NR_Bokforingspost
                                ,NR_Referenssektor
                                ,NR_Referensperiod
                )                                      
                [calc identifier NR_Valuta := "VU4"]                      
                
                *   0                                                                                   
                
                +   sum(
                            Index 
                        
                        group by    NR_Valuta
                                    ,NR_Referensperiod
                    )                                        
                                                                                                                        ; 
//endregion

//region  4.0 UTDATA 
    
    /*  Slutligen lägger jag bara ihop alla dataset till ett utdataset, som kommer användas i nästa
        beräkning: Valutakursomvärderingar */
    
    UTDATA      
            <-  
                sum(
                        union(
                                FIR_index_VU3
                                ,FIR_index_VU2
                                ,FIR_index_VU4
                        )
                        [rename NR_Referenssektor to OLD_Referenssektor]
                        
                        [calc identifier NR_Referenssektor :=
                                    
                                    if OLD_Referenssektor = "S12211" then "S1221"       else
                                    if OLD_Referenssektor = "S12242" then "S12242_1229" else
                                    
                                    OLD_Referenssektor] 
                                    
                    group except OLD_Referenssektor
                )
                                                                                                                        ;        
                
//endregion
